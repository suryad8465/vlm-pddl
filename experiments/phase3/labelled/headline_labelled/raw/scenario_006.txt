(define (problem take_cup_to_living_room)
  (:domain household)

  (:objects
    kitchen - location
    living_room - location
    table - object
    cup - object
  )

  (:init
    (robot-at kitchen)
    (connected kitchen living_room)
    (located table kitchen)
    (located cup kitchen)
    (on cup table)
    (manipulable cup)
  )

  (:goal
    (located cup living_room)
  )
)
